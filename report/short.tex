%% The print settings below (no build label, no provenance footer, black links, page numbers only) are made HERE
%% only, after the shared preamble, so the full and brief versions keep theirs.
\documentclass[paper=a4,fontsize=11pt,DIV=12,parskip=half-,headings=small,abstract=true,
  titlepage=false,bibliography=totoc]{scrartcl}
% A text block of 161.5 x 240 mm (the width of KOMA's DIV=13; the full report's DIV=12 is 157.5 x 222.75 mm), so
% that the report keeps its 16 pages: side margins of about 24 mm, lines about 3 % longer, a top margin of about
% 19 mm and the page number in the bottom margin. Generated figures stay at their natural 6.2-in width. (\areaset,
% unlike DIV=13, does not print typearea's line-length advisory in the log.)
\areaset{161.5mm}{240mm}
% Floats may move past the next \section (as in the brief): a section barrier would force a page break whenever a
% tall figure is still waiting at the end of a short section, leaving half-empty pages. \FloatBarrier is still

% Page furniture for print: page numbers only (no build label in the header, no git or data line in the footer).
\clearpairofpagestyles
\cfoot*{\pagemark}
\pagestyle{scrheadings}
% Run-in paragraph headings with less space above them than KOMA's default (3.25ex): parskip=half- already
% separates the paragraphs and the bold heading marks the start. A little less space between a float and the text.
\RedeclareSectionCommand[beforeskip=.5ex plus .3ex minus .1ex]{paragraph}
\setlength{\textfloatsep}{12pt plus 2pt minus 4pt}
\setlength{\floatsep}{10pt plus 2pt minus 2pt}
% With a ragged bottom, LaTeX's bonus for breaking a page before a heading (\@secpenalty = -300) ends pages several
% lines early before a \section or \paragraph; without it the pages fill (a heading still keeps two lines of its text).
\makeatletter\@secpenalty=0\makeatother

% \needlines{n}: start a new page unless n more lines fit on this one (keeps the References heading with its first
% entries; the logic of needspace's \Needspace*, without loading the package).
\makeatletter
\newcommand{\needlines}[1]{\par\penalty-100\begingroup\dimen@=#1\baselineskip
  \dimen@ii=\pagegoal\advance\dimen@ii-\pagetotal\ifdim\dimen@>\dimen@ii\break\fi\endgroup}
\makeatother
\renewcommand*{\bibfont}{\footnotesize}
\setlength{\bibitemsep}{0pt}
% A compact list for print: no DOIs or optional URLs (an @online entry keeps its URL, an arXiv entry its number) and
% at most three authors per entry.
\ExecuteBibliographyOptions{doi=false,url=false,maxbibnames=3,minbibnames=1}
% The one URL of the list (the W3C specification) breaks only after a '/' or a '.': not inside a word (the

\setcounter{biburlbigbreakpenalty}{10000}
\setcounter{biburllcpenalty}{0}\setcounter{biburlucpenalty}{0}\setcounter{biburlnumpenalty}{0}
% Two entries of the shared references.bib, adjusted for this version only (main and brief are not affected): the
% ISEC 2025 paper's note repeats its date ("Presented 22 February 2025. 2025."), and the W3C specification's note

\DeclareSourcemap{\maps[datatype=bibtex]{\map{\step[fieldsource=entrykey, match=\regexp{singh2025dominant}, final]

  \step[fieldset=note, null]}}}
