% \R{unit/key}: a generated number with its unit sign; \Rb: the bare number; \Rci / \Rcib: with its CI.
% Never use them in headings or captions of floats that move to the list of figures (none is printed here).
\makeatletter
\newcommand{\resmissing}[1]{\textcolor{red!70!black}{\textbf{??\texttt{\detokenize{#1}}}}%
  \PackageWarning{results}{Undefined result `\detokenize{#1}'}}
\NewDocumentCommand{\R}{m}{%
  \ifcsname res@#1\endcsname\csname res@#1\endcsname\else\resmissing{#1}\fi}
\NewDocumentCommand{\Rb}{m}{%
  \ifcsname resb@#1\endcsname\csname resb@#1\endcsname
  \else\ifcsname res@#1\endcsname\csname res@#1\endcsname\else\resmissing{#1}\fi\fi}
\makeatother
\NewDocumentCommand{\Rci}{m}{\R{#1}~[\Rb{#1_lo}, \Rb{#1_hi}]}
\NewDocumentCommand{\Rcib}{m}{\Rb{#1}~[\Rb{#1_lo}, \Rb{#1_hi}]}

% Generated assets. A missing one is a grey box naming the file (a final build stops in the asset step).
\newcommand{\missingasset}[2]{%
  \fcolorbox{black!45}{black!4}{\parbox[c][2.3cm][c]{0.9\linewidth}{\centering\sffamily\small
    \textbf{#1} not generated yet\\[3pt]\scriptsize\expandafter\nolinkurl\expandafter{\detokenize{#2}}}}}
\newsavebox{\fitbox}
% \fitwidth{content}: the content at its natural size, shrunk to the line width only when it is wider.
\newcommand{\fitwidth}[1]{\sbox{\fitbox}{#1}%
  \ifdim\wd\fitbox>\linewidth\resizebox{\linewidth}{!}{\usebox{\fitbox}}\else\usebox{\fitbox}\fi}
% \genfig[includegraphics options]{unit/NAME}: figures are drawn at their final width (6.2 in = the text
% width, or 3.0 in) with 8-pt text, so the default is the natural size, capped at the line width.
\makeatletter
\newcommand{\genfig}[2][]{%
  \IfFileExists{figures/#2.pdf}{\genfig@inc{#1}{figures/#2.pdf}}{%
    \IfFileExists{figures/#2.png}{\genfig@inc{#1}{figures/#2.png}}{%
      \missingasset{#2}{report/figures/#2.pdf}}}}
\newcommand{\genfig@inc}[2]{\ifstrempty{#1}{\fitwidth{\includegraphics{#2}}}{\includegraphics[#1]{#2}}}
\makeatother
% \gentab{unit/name}: the generated file is a complete table float (caption, label tab:<name>).
\newcommand{\gentab}[1]{%
  \IfFileExists{tables/#1.tex}{\input{tables/#1.tex}}{%
